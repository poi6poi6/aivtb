```powershell
Get-ChildItem tests -File | Where-Object { $_.Extension -in '.js', '.cjs' } | ForEach-Object {
    node $_.FullName
    if ($LASTEXITCODE -ne 0) { throw "测试失败：$($_.Name)" }
}
node --test --test-isolation=none minecraft/campaign.test.js minecraft/lifecycle.test.js
.\.venv\Scripts\python.exe -m pip check
```
